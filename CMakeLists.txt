cmake_minimum_required(VERSION 3.20)
project(ictc_native_reproduction LANGUAGES CXX)
set(NS3_ROOT "" CACHE PATH "Installed or built ns-3.46.1 source tree")
if(NOT EXISTS "${NS3_ROOT}/build/include/ns3/core-module.h")
  message(FATAL_ERROR "NS3_ROOT must provide build/include/ns3/core-module.h")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/runtime")
foreach(module core network internet applications point-to-point traffic-control flow-monitor stats)
  find_library(NS3_${module} NAMES ns3.46.1-${module}-optimized ns3.46.1-${module}
               PATHS "${NS3_ROOT}/build/lib" NO_DEFAULT_PATH REQUIRED)
  list(APPEND NS3_LIBRARIES "${NS3_${module}}")
endforeach()
foreach(engine ictc_n12_abilene ictc_n13_transport)
  add_executable(${engine} code/ns3/scratch/${engine}.cc)
  target_include_directories(${engine} PRIVATE "${NS3_ROOT}/build/include")
  target_compile_definitions(${engine} PRIVATE NS3_BUILD_PROFILE_OPTIMIZED)
  target_link_libraries(${engine} PRIVATE ${NS3_LIBRARIES})
endforeach()
